"""Emergent-managed Clerk auth, inert until CLERK_PUBLISHABLE_KEY/CLERK_SECRET_KEY
are set: /api/clerk/config for the frontend, the /api/__clerk production relay
(zero DNS per app), and the require_user dependency for protected routes."""

import os
from functools import lru_cache

import jwt
import requests
from fastapi import APIRouter, HTTPException, Request, Response
from requests.adapters import HTTPAdapter
from starlette.concurrency import run_in_threadpool
from urllib3.util.retry import Retry

CLERK_FRONTEND_API = "https://frontend-api.clerk.dev"
CLERK_BACKEND_API = "https://api.clerk.com/v1"
PROXY_PATH = "/api/__clerk"

# Per-hop headers are never relayed; requests decodes the body, so its encoding goes too.
_HOP_HEADERS = {
    "host", "connection", "keep-alive", "content-length", "transfer-encoding",
    "te", "trailer", "upgrade", "proxy-authorization", "proxy-authenticate",
    "content-encoding",
}

clerk_router = APIRouter(prefix="/api")
_http = requests.Session()
# Clerk's edge closes idle keep-alive connections; retry idempotent calls on a dropped one.
_retry = Retry(total=2, connect=2, read=2, status=0, backoff_factor=0.1,
               allowed_methods=frozenset({"GET", "HEAD", "OPTIONS"}))
_http.mount("https://", HTTPAdapter(pool_maxsize=40, max_retries=_retry))  # matches the threadpool size


def _publishable_key() -> str:
    return os.environ.get("CLERK_PUBLISHABLE_KEY", "")


def _secret_key() -> str:
    return os.environ.get("CLERK_SECRET_KEY", "")


def _is_production() -> bool:
    # Only production instances are served through the /api/__clerk proxy.
    return _publishable_key().startswith("pk_live_")


def _client_ip(request: Request) -> str:
    forwarded = request.headers.get("x-forwarded-for", "")
    if forwarded:
        return forwarded.split(",")[0].strip()
    return request.client.host if request.client else ""


@clerk_router.get("/clerk/config")
def clerk_config():
    """What the frontend needs to start Clerk (the publishable key is public)."""
    key = _publishable_key()
    return {
        "enabled": bool(key and _secret_key()),
        "publishable_key": key,
        "proxy_url": PROXY_PATH if _is_production() else "",
    }


@clerk_router.api_route("/__clerk/{path:path}", methods=["GET", "POST", "PUT", "PATCH", "DELETE"])
async def clerk_proxy(path: str, request: Request):
    """Relay Clerk's Frontend API through this app's own domain."""
    if not _is_production():
        raise HTTPException(status_code=404)
    host = request.headers.get("x-forwarded-host") or request.headers.get("host", "")
    headers = {k: v for k, v in request.headers.items() if k.lower() not in _HOP_HEADERS}
    headers["Clerk-Proxy-Url"] = f"https://{host.split(',')[0].strip()}{PROXY_PATH}"
    headers["Clerk-Secret-Key"] = _secret_key()
    headers["X-Forwarded-For"] = _client_ip(request)
    headers["Accept-Encoding"] = "gzip"  # requests cannot decode brotli

    try:
        upstream = await run_in_threadpool(
            _http.request,
            request.method,
            f"{CLERK_FRONTEND_API}/{path}",
            params=list(request.query_params.multi_items()),
            headers=headers,
            data=await request.body(),
            allow_redirects=False,
            timeout=30,
        )
    except requests.RequestException:
        # Clerk's frontend retries a 502; an unhandled error would be a bare 500.
        return Response(status_code=502)
    response = Response(content=upstream.content, status_code=upstream.status_code)
    raw = upstream.raw.headers
    for name in {n.lower() for n in raw.keys()} - _HOP_HEADERS:
        # getlist keeps every Set-Cookie separate; joining them breaks sessions.
        for value in raw.getlist(name):
            response.raw_headers.append((name.encode(), value.encode()))
    return response


@lru_cache(maxsize=1)
def _jwks_client(secret_key: str) -> jwt.PyJWKClient:
    # Caches Clerk's signing keys and refetches when it meets a new key id.
    return jwt.PyJWKClient(
        f"{CLERK_BACKEND_API}/jwks",
        # Clerk's edge rejects urllib's default User-Agent (Cloudflare 1010).
        headers={"Authorization": f"Bearer {secret_key}", "User-Agent": "emergent-clerk-auth"},
        lifespan=3600,
        timeout=10,
    )


def verify_session_token(token: str) -> dict:
    """Return the verified claims of a Clerk session token, or raise 401."""
    try:
        key = _jwks_client(_secret_key()).get_signing_key_from_jwt(token).key
        return jwt.decode(token, key=key, algorithms=["RS256"], leeway=5)
    except jwt.PyJWTError:
        raise HTTPException(status_code=401, detail="Invalid or expired session")


def require_user(request: Request) -> dict:
    """FastAPI dependency: the signed-in user from `Authorization: Bearer`
    (frontend getToken()) or Clerk's `__session` cookie; user["user_id"] is stable."""
    if not _secret_key():
        raise HTTPException(status_code=503, detail="Clerk is not configured")
    auth = request.headers.get("authorization", "")
    token = auth[7:] if auth.lower().startswith("bearer ") else request.cookies.get("__session", "")
    if not token:
        raise HTTPException(status_code=401, detail="Not signed in")
    claims = verify_session_token(token)
    return {"user_id": claims["sub"], "session_id": claims.get("sid"), "claims": claims}

"""Roles + admin authorization on top of Emergent-managed Clerk sign-in.

Clerk owns identity (passwords, Google, sessions, signed JWTs). The app owns roles:
each person has a `users` doc keyed by the Clerk user id, holding `role`. The role
is read from the database on EVERY request — never from the token or the body.
"""

import hashlib
import logging
import os
import uuid
from datetime import datetime, timedelta, timezone

import requests
from fastapi import Depends, HTTPException
from pymongo import ReturnDocument
from starlette.concurrency import run_in_threadpool

from clerk_auth import require_user
from lib.db import db

logger = logging.getLogger(__name__)

# The role list is code (ships on every publish). Highest first; keep `admin`.
ROLES = ["admin", "citizen"]
DEFAULT_ROLE = "citizen"
RETIRED_ROLES: dict[str, str] = {}  # removed role -> replacement; keep entries
RANK = {r: len(ROLES) - i for i, r in enumerate(ROLES)}

# Which Clerk instance signed the person in (first publish copies preview users to Live).
ENV = "live" if os.environ.get("CLERK_PUBLISHABLE_KEY", "").startswith("pk_live_") else "preview"
IN_ENV = {"env": {"$in": [ENV, None]}}

LAST_SEEN_EVERY = timedelta(minutes=5)


def effective_role(role) -> str:
    role = RETIRED_ROLES.get(role, role)
    return role if role in ROLES else DEFAULT_ROLE


async def move_retired_roles() -> None:
    for old, new in RETIRED_ROLES.items():
        await db.users.update_many({"role": old}, {"$set": {"role": new}})


def _fetch_clerk_user(user_id: str) -> dict:
    r = requests.get(
        f"https://api.clerk.com/v1/users/{user_id}",
        timeout=10,
        headers={"Authorization": f"Bearer {os.environ['CLERK_SECRET_KEY']}", "User-Agent": "emergent-app"},
    )
    r.raise_for_status()
    return r.json()


def _profile(clerk_user: dict) -> tuple[str, str, bool]:
    """(primary email, display name, is-admin-by-verified-ADMIN_EMAIL)."""
    admins = {e.strip().lower() for e in os.environ.get("ADMIN_EMAIL", "").split(",") if e.strip()}
    emails = clerk_user.get("email_addresses") or []
    primary_id = clerk_user.get("primary_email_address_id")
    primary = next((e for e in emails if e.get("id") == primary_id), emails[0] if emails else {})
    email = (primary.get("email_address") or "").lower()
    name = " ".join(p for p in (clerk_user.get("first_name"), clerk_user.get("last_name")) if p) or email
    is_admin = any(
        e["email_address"].lower() in admins and (e.get("verification") or {}).get("status") == "verified"
        for e in emails
    )
    return email, name, is_admin


async def current_member(user: dict = Depends(require_user)) -> dict:
    """The signed-in person with their role, created on first sign-in."""
    uid = user["user_id"]
    now = datetime.now(timezone.utc)
    member = await db.users.find_one({"user_id": uid}, {"_id": 0})
    if member is None or not member.get("role") or not member.get("email"):
        # One Clerk lookup on first sign-in (never per request). Errors propagate so
        # the next request retries instead of saving a wrong role.
        clerk_user = await run_in_threadpool(_fetch_clerk_user, uid)
        email, name, _ = _profile(clerk_user)
        role = DEFAULT_ROLE  # admin rights are NOT stored roles: see require_admin
        await db.users.update_one(
            {"user_id": uid, "role": {"$in": [None, ""]}}, {"$set": {"role": role}}
        )
        member = await db.users.find_one_and_update(
            {"user_id": uid},
            {
                "$setOnInsert": {"user_id": uid, "role": role, "created_at": now},
                "$set": {"email": email, "name": name},
            },
            upsert=True,
            projection={"_id": 0},
            return_document=ReturnDocument.AFTER,
        )
    updates: dict = {}
    if member.get("env") != ENV:
        updates["env"] = ENV
    last_seen = member.get("last_seen")
    if last_seen is not None and last_seen.tzinfo is None:
        last_seen = last_seen.replace(tzinfo=timezone.utc)
    if last_seen is None or now - last_seen > LAST_SEEN_EVERY:
        updates["last_seen"] = now
    if updates:
        await db.users.update_one({"user_id": uid}, {"$set": updates})
    return {**member, **updates, "env": ENV, "role": DEFAULT_ROLE, "session_id": user.get("session_id")}


def require_role(*roles: str):
    async def check(member: dict = Depends(current_member)) -> dict:
        if member["role"] not in roles:
            raise HTTPException(status_code=403, detail="Not allowed")
        return member

    return check


# ---------------------------------------------------------------- single admin
# Exactly ONE admin: the Clerk user whose PRIMARY email equals ADMIN_EMAIL and is
# VERIFIED. Decided from Clerk on every admin call; no stored role can grant it.
PASSCODE_MAX_FAILS = 5
PASSCODE_LOCK = timedelta(minutes=15)


def admin_email() -> str:
    return os.environ.get("ADMIN_EMAIL", "").split(",")[0].strip().lower()


def is_verified_primary_admin(clerk_user: dict) -> bool:
    target = admin_email()
    if not target:
        return False
    pid = clerk_user.get("primary_email_address_id")
    primary = next((e for e in clerk_user.get("email_addresses") or [] if e.get("id") == pid), None)
    return bool(
        primary
        and (primary.get("email_address") or "").strip().lower() == target
        and (primary.get("verification") or {}).get("status") == "verified"
    )


async def is_admin(member: dict) -> bool:
    try:
        clerk_user = await run_in_threadpool(_fetch_clerk_user, member["user_id"])
    except Exception:
        logger.warning("admin check: Clerk lookup failed")
        return False
    return is_verified_primary_admin(clerk_user)


def passcode_hash(value: str | None = None) -> str | None:
    raw = os.environ.get("ADMIN_PASSCODE", "") if value is None else value
    return hashlib.sha256(raw.encode()).hexdigest() if raw else None


async def admin_identity(member: dict = Depends(current_member)) -> dict:
    if not await is_admin(member):
        raise HTTPException(status_code=403, detail="Not allowed")
    return member


async def require_admin(member: dict = Depends(admin_identity)) -> dict:
    """Every admin API: verified ADMIN_EMAIL + (if ADMIN_PASSCODE set) unlocked session."""
    if passcode_hash():
        sid = member.get("session_id") or ""
        if not await db.admin_unlocks.find_one({"_id": sid, "user_id": member["user_id"]}):
            raise HTTPException(status_code=423, detail="Admin passcode required")
    return {**member, "role": "admin"}


async def audit(admin: dict, action: str, target: str) -> None:
    """Append-only log of every admin write — shown on the admin Audit log page."""
    await db.admin_audit.insert_one(
        {
            "id": str(uuid.uuid4()),
            "admin_email": admin.get("email", ""),
            "action": action,
            "target": target,
            "at": datetime.now(timezone.utc),
        }
    )

    for collection, models in INDEXES.items():
        for model in models:  # one at a time so a bad spec skips only itself

=== FILE: backend/lib/doorstep.py ===
"""Doorstep Garbage Vehicle — DEMO ONLY simulation engine.

No background job: the vehicle position is computed from the trip's clock (start time,
speed, pauses) every time a page polls, and pin statuses/alerts are advanced then.
Each pin is alerted/collected at most once per trip (atomic conditional updates).
"""

import math
import uuid
from datetime import datetime, timezone

from lib.db import db

BASE_SPEED_MS = 20_000 / 3600  # simulated truck: 20 km/h, multiplied by the 1x/5x/10x demo speed
COLLECT_M = 60  # vehicle this close to a pin with waste -> collected
ON_ROUTE_M = 150  # pins further than this from the path are never auto-collected
FEED_MAX = 60


def now() -> datetime:
    return datetime.now(timezone.utc)


def iso(dt: datetime | None) -> str:
    if dt is None:
        return ""
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt.isoformat()


def dist_m(a: dict, b: dict) -> float:
    r = 6371000.0
    p1, p2 = math.radians(a["lat"]), math.radians(b["lat"])
    dp, dl = p2 - p1, math.radians(b["lng"] - a["lng"])
    h = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    return 2 * r * math.asin(min(1.0, math.sqrt(h)))


def cumulative(path: list[dict]) -> list[float]:
    out = [0.0]
    for a, b in zip(path, path[1:]):
        out.append(out[-1] + dist_m(a, b))
    return out


def point_at(path: list[dict], cum: list[float], d: float) -> dict:
    if d <= 0:
        return dict(path[0])
    if d >= cum[-1]:
        return dict(path[-1])
    for i in range(1, len(cum)):
        if cum[i] >= d:
            seg = cum[i] - cum[i - 1] or 1
            t = (d - cum[i - 1]) / seg
            a, b = path[i - 1], path[i]
            return {"lat": a["lat"] + (b["lat"] - a["lat"]) * t, "lng": a["lng"] + (b["lng"] - a["lng"]) * t}
    return dict(path[-1])


def project(path: list[dict], cum: list[float], p: dict) -> tuple[float, float]:
    """(distance along path of the closest point, distance from p to the path) in metres."""
    best = (0.0, float("inf"))
    k = math.cos(math.radians(p["lat"]))
    for i in range(1, len(path)):
        a, b = path[i - 1], path[i]
        ax, ay = a["lng"] * k, a["lat"]
        bx, by = b["lng"] * k, b["lat"]
        px, py = p["lng"] * k, p["lat"]
        dx, dy = bx - ax, by - ay
        L2 = dx * dx + dy * dy
        t = 0.0 if L2 == 0 else max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / L2))
        q = {"lat": a["lat"] + (b["lat"] - a["lat"]) * t, "lng": a["lng"] + (b["lng"] - a["lng"]) * t}
        off = dist_m(p, q)
        if off < best[1]:
            best = (cum[i - 1] + (cum[i] - cum[i - 1]) * t, off)
    return best


def speed_ms(trip: dict) -> float:
    return BASE_SPEED_MS * int(trip.get("speed", 1))


def eta_minutes(distance: float, trip: dict) -> int:
    return max(1, math.ceil(distance / speed_ms(trip) / 60))


def progress_m(trip: dict, at: datetime | None = None) -> float:
    """Distance travelled along the path (auto mode) from the trip clock."""
    elapsed = float(trip.get("elapsed_s", 0.0))
    if trip.get("state") == "running" and trip.get("mode") == "auto" and trip.get("resumed_at"):
        resumed = trip["resumed_at"]
        if resumed.tzinfo is None:
            resumed = resumed.replace(tzinfo=timezone.utc)
        elapsed += ((at or now()) - resumed).total_seconds()
    return elapsed * speed_ms(trip)


def vehicle_pos(trip: dict) -> dict:
    if trip.get("mode") == "manual" and trip.get("manual_pos"):
        return trip["manual_pos"]
    return point_at(trip["path"], trip["cum"], progress_m(trip))


def minutes_text(minutes: int) -> str:
    return f"{minutes} minute" if minutes == 1 else f"{minutes} minutes"


def alert_texts(minutes: int) -> tuple[str, str]:
    return (
        f"The garbage vehicle is about {minutes_text(minutes)} away. Please bring your waste down.",
        f"कचरा गाडी सुमारे {minutes} मिनिटांत पोहोचेल. कृपया तुमचा कचरा खाली आणा.",
    )

def pin_status(pin: dict, trip: dict | None) -> str:
    st = ((trip or {}).get("pins") or {}).get(pin["key"], {})
    if st.get("collected_at"):
        return "collected"
    if pin["kind"] == "real":
        rs = pin.get("req_status")
        if rs == "collected":
            return "collected"
        if st.get("alerted_at") or rs == "near":
            return "near"
        return "requested" if rs else "waiting"
    if not pin.get("has_waste"):
        return "waiting"  # nothing to collect today
    return "near" if st.get("alerted_at") else "requested"

async def advance(trip: dict | None) -> dict | None:
    """Move the simulation forward to 'now': alerts, collections, end of trip."""
    if not trip or trip.get("state") not in ("running", "paused"):
        return trip
    pos = vehicle_pos(trip)
    pins = await all_pins()
    state = trip.get("pins") or {}
    auto = trip.get("mode") != "manual"
    travelled = progress_m(trip)
    for pin in pins:
        d = dist_m(pos, pin)
        along, off = project(trip["path"], trip["cum"], pin)
        # the vehicle moves up to ~170 m between polls at 10x: anything on the route it has
        # already passed counts as reached (alert first, then collect)
        passed = auto and off <= ON_ROUTE_M and along <= travelled
        st = state.get(pin["key"], {})
        if pin["alerts_on"] and (d <= pin["alert_m"] or passed) and not st.get("alerted_at") and not st.get("collected_at"):
            res = await db.demo_trips.update_one(
                {"id": trip["id"], f"pins.{pin['key']}.alerted_at": {"$exists": False}},
                {"$set": {f"pins.{pin['key']}.alerted_at": iso(now())}},
            )
            if res.modified_count == 1:
                minutes = eta_minutes(d, trip)
                label = "Real user " + pin["name"] if pin["kind"] == "real" else pin["name"]
                await push_feed(trip["id"], f"{label} alerted at {int(round(d))} m, about {minutes_text(minutes)} away")
                if pin["kind"] == "real":
                    en, mr = alert_texts(minutes)
                    await db.pickup_alerts.insert_one(
                        {
                            "id": str(uuid.uuid4()),
                            "user_id": pin["user_id"],
                            "trip_id": trip["id"],
                            "text_en": en,
                            "text_mr": mr,
                            "distance_m": int(round(d)),
                            "eta_min": minutes,
                            "at": now(),
                        }
                    )
                    await db.pickup_spots.update_one(
                        {"user_id": pin["user_id"], "request.status": "requested"},
                        {"$set": {"request.status": "near"}},
                    )
        if pin["has_waste"] and (d <= COLLECT_M or passed) and not st.get("collected_at"):
            await mark_collected(trip, pin)
    if trip.get("mode") == "auto" and progress_m(trip) >= trip["cum"][-1]:
        await finish_trip(trip, "finished")
    return await db.demo_trips.find_one({"id": trip["id"]}, {"_id": 0})


async def finish_trip(trip: dict, state: str) -> None:
    fresh = await db.demo_trips.find_one({"id": trip["id"]}, {"_id": 0}) or trip
    if fresh.get("state") in ("finished", "stopped"):
        return
    pins_state = fresh.get("pins") or {}
    collected = sum(1 for v in pins_state.values() if v.get("collected_at"))
    alerts = sum(1 for v in pins_state.values() if v.get("alerted_at"))
    started = fresh.get("created_at")
    if started is not None and started.tzinfo is None:
        started = started.replace(tzinfo=timezone.utc)
    travelled = min(progress_m(fresh), fresh["cum"][-1]) if state == "finished" else progress_m(fresh)
    summary = {
        "facility_name": fresh.get("facility_name") or "the drop facility",
        "distance_km": round(min(travelled, fresh["cum"][-1]) / 1000, 2),
        "minutes": max(1, int(((now() - started).total_seconds() if started else 0) // 60)),
        "collected": collected,
        "alerts": alerts,
        "stops": max(0, len(fresh.get("path", [])) - 2),
    }
    res = await db.demo_trips.update_one(
        {"id": fresh["id"], "state": {"$in": ["running", "paused"]}},
        {
            "$set": {
                "state": state,
                "elapsed_s": progress_m(fresh) / speed_ms(fresh),
                "resumed_at": None,
                "summary": summary,
                "ended_at": now(),
            }
        },
    )
    if res.modified_count == 1:
        if state == "finished":
            await push_feed(fresh["id"], f"Trip complete. Dropped at facility {summary['facility_name']}.")
        else:
            await push_feed(fresh["id"], "Trip stopped by admin.")

"""Google Gemini (free tier) — the single place the app talks to the AI.

Key:   GEMINI_API_KEY  (backend .env only, never in code / logs / git)
Model: GEMINI_MODEL    (default gemini-3.5-flash — free tier, accepts images)
Every failure becomes an AIError with a friendly, user-facing message.
"""

import asyncio
import logging
import os

import httpx
from google import genai
from google.genai import errors, types

logger = logging.getLogger(__name__)

DEFAULT_MODEL = "gemini-3.5-flash"

FRIENDLY = {
    "busy": "AI is busy, please try again in a minute.",
    "key": "The AI service is not set up correctly (check GEMINI_API_KEY in backend/.env).",
    "offline": "Cannot reach the AI service. Check your internet connection and try again.",
    "invalid": "The AI could not read this clearly. Please retake a sharper, well-lit photo or add a short description.",
    "error": "The AI service had a problem. Please try again.",
}
HTTP_STATUS = {"busy": 429, "key": 503, "offline": 503, "invalid": 422, "error": 502}


class AIError(Exception):
    def __init__(self, kind: str):
        self.kind = kind
        self.message = FRIENDLY[kind]
        self.status = HTTP_STATUS[kind]
        super().__init__(self.message)


_client: genai.Client | None = None
_client_key: str | None = None


def model_name() -> str:
    return os.environ.get("GEMINI_MODEL", "").strip() or DEFAULT_MODEL


def fallback_model() -> str:
    return os.environ.get("GEMINI_FALLBACK_MODEL", "").strip()


def is_configured() -> bool:
    return bool(os.environ.get("GEMINI_API_KEY", "").strip())


def client() -> genai.Client:
    global _client, _client_key
    key = os.environ.get("GEMINI_API_KEY", "").strip()
    if not key:
        raise AIError("key")
    if _client is None or key != _client_key:
        _client, _client_key = genai.Client(api_key=key), key
    return _client


async def generate(contents, config: types.GenerateContentConfig):
    """Gemini call with one automatic retry when the free tier is busy (429/503)."""
    try:
        return await _generate_once(contents, config, model_name())
    except AIError as exc:
        if exc.kind != "busy":
            raise
        backup = fallback_model()
        if backup and backup != model_name():
            # free-tier quota of the main model used up -> switch to the backup model
            return await _generate_once(contents, config, backup)
        await asyncio.sleep(3)
        return await _generate_once(contents, config, model_name())


async def _generate_once(contents, config: types.GenerateContentConfig, model: str):
    """One Gemini call; maps SDK/network errors to AIError (never leaks the key)."""
    try:
        return await client().aio.models.generate_content(model=model, contents=contents, config=config)
    except AIError:
        raise
    except errors.ClientError as exc:
        text = str(exc).upper()
        if exc.code in (401, 403) or "API_KEY" in text or "API KEY" in text:
            logger.warning("Gemini rejected the API key (code %s)", exc.code)
            raise AIError("key") from None
        if exc.code == 429 or "RESOURCE_EXHAUSTED" in text:
            logger.warning("Gemini free-tier rate limit reached")
            raise AIError("busy") from None
        if exc.code == 400 and ("IMAGE" in text or "INLINE" in text or "MIME" in text):
            raise AIError("invalid") from None
        if exc.code == 404:
            logger.warning("Gemini model %s not found — check GEMINI_MODEL", model_name())
            raise AIError("key") from None
        logger.warning("Gemini client error %s", exc.code)
        raise AIError("error") from None
    except errors.ServerError as exc:
        logger.warning("Gemini server error %s", exc.code)
        raise AIError("busy") from None
    except (httpx.HTTPError, TimeoutError, OSError):
        logger.warning("Gemini unreachable (network)")
        raise AIError("offline") from None
    except errors.APIError as exc:
        logger.warning("Gemini API error %s", getattr(exc, "code", "?"))
        raise AIError("error") from None

"""Basic API security: per-user rate limits for the AI endpoints + secure headers."""

import time
from collections import defaultdict, deque

from fastapi import Depends, HTTPException

from lib.auth import current_member

_hits: dict[str, deque] = defaultdict(deque)


def rate_limit(bucket: str, limit: int, per_seconds: int = 60):
    """FastAPI dependency: at most `limit` calls per user per window (in-memory)."""

    async def check(member: dict = Depends(current_member)) -> dict:
        key = f"{bucket}:{member['user_id']}"
        now = time.monotonic()
        q = _hits[key]
        while q and now - q[0] > per_seconds:
            q.popleft()
        if len(q) >= limit:
            raise HTTPException(status_code=429, detail="Too many requests — please wait a minute and try again.")
        q.append(now)
        return member

    return check


SECURE_HEADERS = {
    "X-Content-Type-Options": "nosniff",
    "X-Frame-Options": "DENY",
    "Referrer-Policy": "strict-origin-when-cross-origin",
    "Permissions-Policy": "camera=(self), geolocation=(self), microphone=()",
}


async def secure_headers_middleware(request, call_next):
    response = await call_next(request)
    for k, v in SECURE_HEADERS.items():
        response.headers.setdefault(k, v)
    return response

@router.get("/admin/gate", response_model=AdminGate)
async def admin_gate(member: dict = Depends(admin_identity)):
    attempts = await db.admin_passcode_attempts.find_one({"_id": member["user_id"]}) or {}
    locked = _aware(attempts.get("locked_until"))
    unlocked = bool(await db.admin_unlocks.find_one({"_id": member.get("session_id") or "", "user_id": member["user_id"]}))
    return AdminGate(
        admin_email=admin_email(),

@router.post("/admin/unlock", response_model=AdminGate)
async def admin_unlock(body: PasscodeIn, member: dict = Depends(admin_identity)):
    import hmac

    now = datetime.now(timezone.utc)
    attempts = await db.admin_passcode_attempts.find_one({"_id": member["user_id"]}) or {}
    locked = _aware(attempts.get("locked_until"))
    if locked and locked > now:
        raise HTTPException(status_code=429, detail="Too many wrong attempts. Try again in 15 minutes.")
    expected = passcode_hash()
    if expected and not hmac.compare_digest(passcode_hash(body.passcode) or "", expected):
        fails = attempts.get("fails", 0) + 1
        update = {"fails": fails}
        if fails >= PASSCODE_MAX_FAILS:
            update = {"fails": 0, "locked_until": now + PASSCODE_LOCK}
        await db.admin_passcode_attempts.update_one({"_id": member["user_id"]}, {"$set": update}, upsert=True)
        left = PASSCODE_MAX_FAILS - fails
        raise HTTPException(status_code=401, detail="Wrong passcode." + (f" {left} attempt(s) left." if left > 0 else " Locked for 15 minutes."))
    await db.admin_passcode_attempts.delete_one({"_id": member["user_id"]})
    await db.admin_unlocks.update_one(
        {"_id": member.get("session_id") or ""}, {"$set": {"user_id": member["user_id"], "at": now}}, upsert=True
    )
    await audit(member, "admin_unlock", "admin panel unlocked")
    return AdminGate(admin_email=admin_email(), passcode_required=bool(expected), unlocked=True, locked_until=None)

@router.get("/admin/stats", response_model=AdminStats)
async def admin_stats(admin: dict = Depends(require_admin)):
    now = datetime.now(timezone.utc)
    since = (now - timedelta(days=13)).replace(hour=0, minute=0, second=0, microsecond=0)

    by_category: dict[str, int] = {}
    async for row in db.scans.aggregate([{"$group": {"_id": "$category", "n": {"$sum": 1}}}]):
        by_category[row["_id"] or "general"] = row["n"]

    points = 0
    async for row in db.scans.aggregate([{"$group": {"_id": None, "p": {"$sum": "$eco_points"}}}]):
        points = row["p"]

    daily_map: dict[str, int] = {}
    async for row in db.scans.aggregate(
        [
            {"$match": {"created_at": {"$gte": since}}},
            {"$group": {"_id": {"$dateToString": {"format": "%Y-%m-%d", "date": "$created_at"}}, "n": {"$sum": 1}}},
        ]
    ):
        daily_map[row["_id"]] = row["n"]
    daily = [
        DayCount(date=(since + timedelta(days=i)).strftime("%Y-%m-%d"), count=daily_map.get((since + timedelta(days=i)).strftime("%Y-%m-%d"), 0))
        for i in range(14)
    ]

    top_items = [
        ItemCount(item_name=row["_id"], count=row["n"])
        async for row in db.scans.aggregate(
            [{"$group": {"_id": "$item_name", "n": {"$sum": 1}}}, {"$sort": {"n": -1}}, {"$limit": 6}]
        )
    ]

    recent: list[ActivityItem] = []
    emails = {u["user_id"]: u.get("email", "") async for u in db.users.find({}, {"_id": 0, "user_id": 1, "email": 1})}
    async for s in db.scans.find({}, {"_id": 0, "item_name": 1, "category": 1, "user_id": 1, "created_at": 1}).sort("created_at", -1).limit(8):
        who = emails.get(s.get("user_id") or "", "") or "demo (before sign-in)"
        recent.append(ActivityItem(kind="scan", text=f"{who} scanned {s['item_name']} ({s['category']})", at=_aware(s["created_at"])))
    async for f in db.feedback.find({}, {"_id": 0}).sort("created_at", -1).limit(5):
        recent.append(ActivityItem(kind="feedback", text=f"{f.get('user_email', '')} sent {f['kind'].replace('_', ' ')}: {f['message'][:60]}", at=_aware(f["created_at"])))
    async for u in db.users.find({**IN_ENV, "created_at": {"$ne": None}}, {"_id": 0}).sort("created_at", -1).limit(5):
        recent.append(ActivityItem(kind="signup", text=f"{u.get('email', '')} joined", at=_aware(u["created_at"])))
    recent.sort(key=lambda a: a.at, reverse=True)

    return AdminStats(
        total_users=await db.users.count_documents(IN_ENV),
        total_scans=await db.scans.count_documents({}),
        total_eco_points=points,
        total_feedback=await db.feedback.count_documents({}),
        open_feedback=await db.feedback.count_documents({"status": {"$ne": "resolved"}}),
        active_facilities=await db.facilities.count_documents({"active": {"$ne": False}}),
        total_facilities=await db.facilities.count_documents({}),
        low_confidence_scans=await db.scans.count_documents({"confidence": {"$lt": 0.65}}),
        by_category=by_category,
        daily=daily,
        top_items=top_items,
        recent=recent[:12],
    )

@router.get("/admin/scans", response_model=ScanPage)
async def admin_scans(

    category: str = "",
    user_id: str = "",
    low_confidence: bool = False,
    days: int = Query(default=0, ge=0, le=3650),
    page: int = Query(default=1, ge=1),

    query: dict = {}
    if q.strip():
        rx = {"$regex": re.escape(q.strip()), "$options": "i"}

    if user_id:

        query["confidence"] = {"$lt": 0.65}
    if days:
        query["created_at"] = {"$gte": datetime.now(timezone.utc) - timedelta(days=days)}

    total = await db.scans.count_documents(query)
    docs = (
        await db.scans.find(query, {"_id": 0})
        .sort("created_at", -1)
        .skip((page - 1) * page_size)
        .limit(page_size)
        .to_list(page_size)
    )
    ids = list({d.get("user_id") for d in docs if d.get("user_id")})
    emails = {u["user_id"]: u.get("email", "") async for u in db.users.find({"user_id": {"$in": ids}}, {"_id": 0})}
    items = []
    for d in docs:
        d["created_at"] = _aware(d["created_at"])
        items.append(AdminScan(**d, user_email=emails.get(d.get("user_id") or "", "")))
    return ScanPage(items=items, total=total, page=page, page_size=page_size)

def extract_json(text: str) -> dict:
    t = text.strip()
    start, end = t.find("{"), t.rfind("}")
    if start == -1 or end <= start:
        raise ValueError(f"no JSON object in model reply: {text[:200]!r}")
    return json.loads(t[start : end + 1])


def guide_text(guidelines: list[dict]) -> str:
    lines = []
    for g in guidelines:
        line = f"- {g['key']} ({g['name']}): {g.get('description', '')}"
        if g.get("examples"):
            line += f" Examples: {', '.join(g['examples'])}."
        if g.get("disposal_steps"):
            line += f" Official local disposal steps: {' | '.join(g['disposal_steps'])}."
        if g.get("hazards"):
            line += f" Known hazards: {' | '.join(g['hazards'])}."
        lines.append(line)
    return "\n".join(lines)


@router.post("/classify", response_model=Classification)
async def classify_item(input: ClassificationRequest, member: dict = Depends(rate_limit("classify", 15))):
    if not input.image_base64 and not (input.hint and input.hint.strip()):
        raise HTTPException(status_code=422, detail="Provide an image or a text hint describing the waste item.")
    if input.image_base64 and len(input.image_base64) > 9_000_000:
        raise HTTPException(status_code=413, detail="Image too large — retake or upload a smaller photo.")

    guidelines = await list_guidelines(active_only=True)
    by_key = {g["key"]: g for g in guidelines}
    categories = list(by_key) or CATEGORIES
    system = SYSTEM_TEMPLATE.format(
        cats=", ".join(categories), language=LANGUAGE_NAMES[input.language]
    ).replace("<<GUIDE>>", guide_text(guidelines))

    desc = []
    if input.hint and input.hint.strip():
        desc.append(f"User's description / hint: {input.hint.strip()}")
    if input.image_base64:
        desc.append("Identify the single waste item shown in the attached photo.")
    text = "\n".join(desc)

    contents: list = [types.Part.from_text(text=text)]
    if input.image_base64:
        if input.mime_type not in ALLOWED_MIME:
            raise HTTPException(status_code=415, detail="Only JPEG, PNG or WEBP photos are accepted.")
        try:
            image_bytes = base64.b64decode(input.image_base64, validate=True)
        except (binascii.Error, ValueError):
            raise HTTPException(status_code=422, detail="The photo could not be read — please retake it.")
        contents.insert(0, types.Part.from_bytes(data=image_bytes, mime_type=input.mime_type))

    config = types.GenerateContentConfig(
        system_instruction=system,
        temperature=0.2,
        response_mime_type="application/json",
        response_schema=response_schema(categories),
    )
    data = None
    try:
        for _attempt in range(2):  # validate every result; retry once if invalid
            reply = await generate(contents, config)
            try:
                candidate = extract_json(reply.text or "")
                if candidate.get("category") in categories and 0 <= float(candidate.get("confidence")) <= 1:
                    data = candidate
                    break
            except (ValueError, TypeError):
                pass
            logger.warning("classify: invalid AI result, retrying once")
    except AIError as exc:
        raise HTTPException(status_code=exc.status, detail=exc.message) from None
    if data is None:
        raise HTTPException(status_code=422, detail=FRIENDLY["invalid"])

    category = data.get("category", "general")
    if category not in categories:
        category = "general"
    guideline = by_key.get(category, {})
    try:
        confidence = max(0.0, min(1.0, float(data.get("confidence", 0.5))))
    except (TypeError, ValueError):
        confidence = 0.5

    result = Classification(
        item_name=str(data.get("item_name", "Unknown item"))[:120],
        category=category,
        subcategory=str(data.get("subcategory", ""))[:120],
        confidence=confidence,
        disposal_steps=[str(s) for s in (data.get("disposal_steps") or [])][:8],
        hazards=[str(s) for s in (data.get("hazards") or [])][:6],
        eco_tip=str(data.get("eco_tip", "")),
        eco_points=int(guideline.get("eco_points", ECO_POINTS.get(category, 5))),
        needs_follow_up=bool(data.get("needs_follow_up", False)) and confidence < 0.8,
        follow_up_question=str(data.get("follow_up_question", "")),
        follow_up_options=[str(s) for s in (data.get("follow_up_options") or [])][:3],
        language=input.language,
        hint=input.hint,
        created_at=datetime.now(timezone.utc),
        category_label=str(guideline.get("name", "")),
        user_id=member["user_id"],
    )

    await db.scans.insert_one({"_id": result.id, **result.model_dump()})
    return result

"""GET /api/facilities — category-based facility finder (curated Chiplun seed + live OpenStreetMap)."""

import logging
import math
from datetime import datetime, timedelta, timezone

import httpx
from fastapi import APIRouter, Depends

from lib.auth import current_member
from lib.db import db
from models.waste import Facility

router = APIRouter()
logger = logging.getLogger(__name__)

CHIPLUN_LAT = 17.5323
CHIPLUN_LNG = 73.5186
SEARCH_RADIUS_M = 25_000
CACHE_TTL = timedelta(hours=24)

OVERPASS_URL = "https://overpass-api.de/api/interpreter"
OVERPASS_QUERY = """[out:json][timeout:15];
(
  node["amenity"="recycling"](around:{radius},{lat},{lng});
  way["amenity"="recycling"](around:{radius},{lat},{lng});
  node["shop"="second_hand"](around:{radius},{lat},{lng});
  node["name"~"scrap|kabadi|kabaadi|recycl|e-?waste",i](around:{radius},{lat},{lng});
);
out center tags 40;"""


def haversine_m(lat1: float, lng1: float, lat2: float, lng2: float) -> float:
    r = 6371000.0
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dp = p2 - p1
    dl = math.radians(lng2 - lng1)
    a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    return 2 * r * math.asin(math.sqrt(a))


def osm_categories(tags: dict, name: str) -> list[str]:
    cats: list[str] = []
    blob = " ".join(str(v) for v in tags.values()).lower() + " " + name.lower()
    if (
        tags.get("recycling:e_waste") == "yes"
        or tags.get("recycling:electronics") == "yes"
        or tags.get("recycling:batteries") == "yes"
        or "e-waste" in blob
        or "ewaste" in blob
    ):
        cats.append("ewaste")
    if tags.get("shop") == "second_hand" or "scrap" in blob or "kabadi" in blob:
        cats.extend(["recyclable", "ewaste"])
    if any(tags.get(f"recycling:{m}") == "yes" for m in ("glass", "paper", "cardboard", "metal", "plastic", "cans")):
        cats.append("recyclable")
    if not cats:
        cats.append("general")
    seen: set[str] = set()
    return [c for c in cats if not (c in seen or seen.add(c))]


def map_osm_element(el: dict) -> dict | None:
    tags = el.get("tags", {})
    name = tags.get("name") or "Recycling / collection point"
    loc = el.get("center") or el
    lat, lng = loc.get("lat"), loc.get("lon")
    if lat is None or lng is None:
        return None
    address_parts = [
        tags.get(f"addr:{k}") for k in ("house_number", "street", "suburb", "city", "town", "village", "postcode")
    ]
    address = ", ".join(p for p in address_parts if p) or "Mapped on OpenStreetMap"
    return {
        "id": f"osm-{el.get('type', 'n')}-{el.get('id', 0)}",
        "name": name,
        "categories": osm_categories(tags, name),
        "lat": float(lat),
        "lng": float(lng),
        "address": address,
        "phone": tags.get("phone") or tags.get("contact:phone"),
        "hours": tags.get("opening_hours"),
        "accepts": [v.replace("recycling:", "") for k, v in tags.items() if k.startswith("recycling:") and v == "yes"],
        "source": "osm",
    }


async def refresh_osm(lat: float, lng: float) -> None:
    """Pull live OSM points around (lat, lng); no-op if the cache is fresh or OSM is unreachable."""
    latest = await db.facilities.find_one({"source": "osm"}, sort=[("fetched_at", -1)])
    if latest and latest.get("fetched_at"):
        fetched = latest["fetched_at"]
        if fetched.tzinfo is None:
            fetched = fetched.replace(tzinfo=timezone.utc)
        if datetime.now(timezone.utc) - fetched < CACHE_TTL:
            return
    try:
        query = OVERPASS_QUERY.format(radius=SEARCH_RADIUS_M, lat=lat, lng=lng)
        async with httpx.AsyncClient(timeout=15.0) as client:
            res = await client.post(OVERPASS_URL, content=f"data={query}", headers={"Content-Type": "text/plain"})
            res.raise_for_status()
            elements = res.json().get("elements", [])
        if docs:
            now = datetime.now(timezone.utc)
            for el in elements:
                doc = map_osm_element(el)
                if doc:
                    doc["fetched_at"] = now
                    # upsert by id — never a bulk wipe of the collection in app code
                    # $set keeps an admin's activate/deactivate choice on refreshed points
                    await db.facilities.update_one({"id": doc["id"]}, {"$set": doc}, upsert=True)
            # evict only stale OSM cache entries older than a week (narrow, age-filtered)
            stale_cutoff = now - timedelta(days=7)
            await db.facilities.delete_many({"source": "osm", "fetched_at": {"$lt": stale_cutoff}})
            logger.info("refresh_osm: stored %d OSM facilities", len(docs))
    except Exception as exc:
        logger.warning("refresh_osm failed (serving seed data only): %s", exc)


async def find_facilities(
    category: str | None = None,
    lat: float | None = None,
    lng: float | None = None,
    live: bool = True,
) -> list[dict]:
    lat = CHIPLUN_LAT if lat is None else lat
    lng = CHIPLUN_LNG if lng is None else lng
    if live:
        await refresh_osm(lat, lng)

    # deactivated facilities (admin panel) are hidden from citizens and the chat agent
    docs = await db.facilities.find({"active": {"$ne": False}}, {"_id": 0}).to_list(500)
    out = []
    for doc in docs:
        if category and category not in doc.get("categories", []):
            continue
        doc["distance_m"] = round(haversine_m(lat, lng, doc["lat"], doc["lng"]))
        out.append(doc)
    out.sort(key=lambda d: d["distance_m"])
    return out[:40]

"""

from models.waste import GateRequest, GateStatus

COOKIE_VALUE = "unlocked"

@router.post("/gate/unlock", response_model=GateStatus)

    expected = os.environ.get("GATE_PIN", DEFAULT_PIN)

    return GateStatus(unlocked=True)

    response.delete_cookie(COOKIE_NAME)
    return GateStatus(unlocked=False)

def _out(doc: dict) -> Tip:

        raise HTTPException(status_code=404, detail="Tip not found")

def fetch_b64(url: str) -> str:
    req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
    raw = urllib.request.urlopen(req, timeout=30).read()
    im = Image.open(io.BytesIO(raw)).convert("RGB")
    im.thumbnail((1024, 1024))
    buf = io.BytesIO()
    im.save(buf, "JPEG", quality=85)
    return base64.b64encode(buf.getvalue()).decode()


def post(path: str, body: dict, token: str) -> tuple[int, dict]:
    req = urllib.request.Request(
        API + path,
        data=json.dumps(body).encode(),
        method="POST",
        headers={"Content-Type": "application/json", "Authorization": "Bearer " + token},
    )
    try:
        res = urllib.request.urlopen(req, timeout=120)
        return res.status, json.load(res)
    except urllib.error.HTTPError as e:
        return e.code, json.loads(e.read() or b"{}")
